\section{Introduction}
% ============================================================

% Slide 3: Project Overview
\begin{frame}{Project Overview}
  \begin{block}{Research Goal}
    Compare transformer encoder architectures on multilingual political and
    fake news classification using \textbf{LoRA} (Low-Rank Adaptation)
    parameter-efficient fine-tuning.
  \end{block}

  \vspace{0.3cm}
  \textbf{Key aspects:}
  \begin{itemize}
    \item Four encoder models evaluated across five English-language datasets
    \item Two task types: binary classification and multi-class classification
    \item Metrics: \textbf{Accuracy} and \textbf{Weighted F1-Score}
    \item Each experiment averaged over \textbf{5 runs} for reliability
  \end{itemize}
\end{frame}

\section{Key Findings}
% ============================================================

% Slide 9: Key Findings (1)
\begin{frame}{Key Findings (1/2)}
  \begin{block}{1.\ RoBERTa-large is the overall best model}
    Achieves the highest scores on \textbf{4 out of 5} datasets (Hyperpartisan, All Sides, Fake News Net, CLEF 3A). Scaling from base to large yields consistent improvements of \textbf{+3--6\%} on most tasks.
  \end{block}

  \begin{block}{2.\ POLITICS benefits from domain-specific pre-training}
    Despite being a base-sized model, \texttt{launch/POLITICS} ranks \textbf{2nd overall} and narrows the gap with roberta-large on political tasks:
    \begin{itemize}
      \item All Sides: 0.694 Acc vs.\ roberta-base's 0.652 (\textbf{+4.1\%})
      \item Hyperpartisan: 0.836 Acc vs.\ roberta-base's 0.795 (\textbf{+4.1\%})
    \end{itemize}
    This confirms that \textbf{domain-specific pre-training} provides meaningful gains.
  \end{block}
\end{frame}

% Slide 10: Key Findings (2)
\begin{frame}{Key Findings (2/2)}
  \begin{block}{3.\ ModernBERT-base underperforms expectations}
    Lowest scores on 4 of 5 datasets despite being a newer architecture. Potential causes:
    \begin{itemize}
      \item LoRA target modules (\texttt{Wqkv}) may need further tuning
      \item Architecture novelty may not translate to this task domain
    \end{itemize}
  \end{block}

  \begin{block}{4.\ Multi-class tasks are significantly harder}
    All Sides (3-class) has the lowest scores across all models (best: 0.714 Acc). Binary tasks (Fake News Net, SemEval) are substantially easier, with most models exceeding 0.84 accuracy.
  \end{block}

  \begin{block}{5.\ SemEval 2019 anomaly}
    RoBERTa-base \textbf{outperforms} roberta-large on SemEval 2019 (0.868 vs.\ 0.856). The small dataset size ($\sim$1.3K) may cause the larger model to overfit.
  \end{block}
\end{frame}

\section{Future Work}
% ============================================================

% Slide 11: Future Work
\begin{frame}{Future Work \& Potential Improvements}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Expand Model Coverage:}
      \begin{itemize}
        \item Add \textbf{decoder models}: Llama-3.1-8B, Mistral-Nemo, Qwen2.5-7B with LoRA SFT
        \item Compare encoder SFT vs.\ decoder SFT
        \item Evaluate \textbf{in-context learning}: zero-shot, chain-of-thought, and few-shot prompting
      \end{itemize}

      \vspace{0.3cm}
      \textbf{Multilingual Evaluation:}
      \begin{itemize}
        \item Test on remaining datasets: Fake.br (Portuguese), FakeNewsCorpus (Spanish), CLEF 1C (Bulgarian, Arabic)
        \item Use multilingual encoders: XLM-RoBERTa, mDeBERTa
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Methodological Improvements:}
      \begin{itemize}
        \item Hyperparameter search for LoRA rank, alpha, and target modules (especially for ModernBERT)
        \item Report confidence intervals / standard deviations from the 5 runs
        \item Investigate the SemEval anomaly with per-run analysis
      \end{itemize}

      \vspace{0.3cm}
      \textbf{Additional Analyses:}
      \begin{itemize}
        \item Confusion matrix analysis for multi-class tasks
        \item Effect of dataset size on model performance
        \item Computational cost comparison across models
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
